<script>
  if ("serviceWorker" in navigator) {
    navigator.serviceWorker.register("/service-worker.js")
      .then(() => console.log(" Service Worker regisztrálva"))
      .catch((error) => console.error(" Service Worker hiba:", error));
  }
  let deferredPrompt;
  
  window.addEventListener("beforeinstallprompt", (event) => {
    event.preventDefault();
    deferredPrompt = event;

    // Gomb megjelenítése
    const installBtn = document.createElement("button");
    installBtn.innerText = "Telepítés";
    installBtn.style.position = "fixed";
    installBtn.style.bottom = "20px";
    installBtn.style.right = "20px";
    installBtn.style.padding = "10px 20px";
    installBtn.style.background = "#007bff";
    installBtn.style.color = "#fff";
    installBtn.style.border = "none";
    installBtn.style.borderRadius = "5px";
    installBtn.style.cursor = "pointer";

    document.body.appendChild(installBtn);

    installBtn.addEventListener("click", () => {
      deferredPrompt.prompt();
      deferredPrompt.userChoice.then((choiceResult) => {
        if (choiceResult.outcome === "accepted") {
          console.log("Felhasználó telepítette az appot");
        } else {
          console.log("Felhasználó elutasította a telepítést");
        }
        installBtn.remove();
      });
    });
  });
</script>
